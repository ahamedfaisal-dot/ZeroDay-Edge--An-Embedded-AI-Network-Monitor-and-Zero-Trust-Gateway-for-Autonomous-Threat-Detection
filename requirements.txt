# ZeroDay-Edge — Python Dependencies
# Install: pip install -r requirements.txt
#
# Tuned for Raspberry Pi 4 Model B (4GB RAM). Deep-learning inference runs
# on tflite-runtime instead of full TensorFlow — full TF's ~400MB+ RSS and
# slow import are a Pi 5/8GB thing, not a Pi 4/4GB thing.
#
# One-time setup before deploying to the Pi 4:
#   1. On a dev machine (or the original Pi 5), `pip install tensorflow`
#      and run `python convert_to_tflite.py` to produce *.tflite files.
#   2. Copy ml_models/*.tflite to the Pi 4. Full TensorFlow is NOT needed
#      on the Pi 4 itself — only tflite-runtime below.
# If no *.tflite files are present, ml_engine.py falls back to loading the
# .h5 files directly, which DOES require `tensorflow` installed on-device —
# avoid this on a 4GB Pi 4 if at all possible.

flask>=3.0,<4.0
flask-cors>=4.0,<5.0

# Packet capture (requires: sudo apt install libpcap-dev)
scapy>=2.5,<3.0

# ML — same as main project backend
xgboost>=2.1,<3.0
scikit-learn>=1.5,<2.0
joblib>=1.4,<2.0
numpy>=1.26,<3.0

# Deep learning inference (lightweight — see conversion note above).
# On Raspberry Pi OS, install via piwheels: pip install tflite-runtime
# If wheels aren't available for your Python version/architecture, fall back to:
#   pip install tensorflow-aarch64   (64-bit Pi OS only — will NOT install on
#                                      32-bit armhf/armv7l, use plain
#                                      `tensorflow` there and expect it to be slow)
tflite-runtime>=2.14,<3.0; platform_machine == "aarch64" or platform_machine == "armv7l"

# NOTE: shap is intentionally NOT used. Stage-1 XAI uses xgboost's own
# built-in pred_contribs=True (exact TreeSHAP, computed in its C++ core) —
# no separate package needed. This also sidesteps shap's numba -> llvmlite
# dependency, which has no prebuilt wheel on 32-bit ARM (armv7l) and fails
# trying to compile LLVM from source.

# Threat intel enrichment
requests>=2.31,<3.0

# System info
psutil>=5.9,<7.0
